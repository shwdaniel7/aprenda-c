Ponteiro para função: um endereço que você chama

Um ponteiro para função guarda o endereço de uma função — você pode guardá-lo em variável, passá-lo como argumento e chamá-lo através do parâmetro. É o mecanismo dos callbacks: uma função recebe "o que fazer" como parâmetro.

A sintaxe é esquisita: int (*f)(int) é um ponteiro para função que recebe int e devolve int. Chame com f(x) ou (*f)(x).

Código — fnptr.c
#include <stdio.h>

int dobrar(int v)    { return v * 2; }
int triplicar(int v) { return v * 3; }

int aplicar(int (*f)(int), int valor)
{
    return f(valor);
}

int main(void)
{
    int (*ptr)(int) = dobrar;

    printf("dobrar(5) via ponteiro = %d\n", ptr(5));
    printf("aplicar(triplicar, 5)  = %d\n", aplicar(triplicar, 5));
    printf("aplicar(triplicar, 7)  = %d\n", aplicar(triplicar, 7));
    return 0;
}
Saída esperada
dobrar(5) via ponteiro = 10
aplicar(triplicar, 5)  = 15
aplicar(triplicar, 7)  = 21

qsort e seu comparador

qsort ordena qualquer array, mas não sabe o que significa "maior/menor" para os seus dados — por isso pede um comparador: uma função int cmp(const void *, const void *) que devolve negativo/zero/positivo. O parâmetro é void *; dentro, convertemos e então lemos.

Código — qsort.c
#include <stdio.h>
#include <stdlib.h>

int cmp_int(const void *a, const void *b)
{
    int x = *(const int *)a;
    int y = *(const int *)b;
    return (x > y) - (x < y);
}

int main(void)
{
    int v[] = {42, 7, 19, 3, 88, 1};
    int n = (int)(sizeof v / sizeof v[0]);

    qsort(v, (size_t)n, sizeof v[0], cmp_int);

    for (int i = 0; i < n; i++) {
        printf("%d ", v[i]);
    }
    printf("\n");
    return 0;
}
Saída esperada
1 3 7 19 42 88

typedef para ponteiro de função e tabelas

Uma linha inteira complicada fica legível com typedef: typedef int (*Operacao)(int, int);. Com isso, declara-se um array de funções — a base de menus, máquinas de estado e despachos:

Código — tabela.c
#include <stdio.h>

typedef int (*Operacao)(int, int);

int somar(int a, int b)       { return a + b; }
int subtrair(int a, int b)    { return a - b; }
int multiplicar(int a, int b) { return a * b; }

int main(void)
{
    Operacao ops[3] = {somar, subtrair, multiplicar};
    int a = 12, b = 4;

    for (int i = 0; i < 3; i++) {
        printf("op[%d](%d, %d) = %d\n", i, a, b, ops[i](a, b));
    }
    return 0;
}
Saída esperada
op[0](12, 4) = 16
op[1](12, 4) = 8
op[2](12, 4) = 48

Ponteiro para ponteiro: trocar ponteiros

Ponteiro para ponteiro (int **) é "o endereço de um ponteiro". Serve para funções que modificam o ponteiro do chamador — como já vimos no swap, mas desta vez trocando alvos:

Código — ptrptr.c
#include <stdio.h>

void trocar_ponteiros(int **a, int **b)
{
    int *tmp = *a;
    *a = *b;
    *b = tmp;
}

int main(void)
{
    int x = 1, y = 2;
    int *px = &x, *py = &y;

    printf("antes : px -> %d , py -> %d\n", *px, *py);
    trocar_ponteiros(&px, &py);
    printf("depois: px -> %d , py -> %d\n", *px, *py);
    return 0;
}
Saída esperada
antes : px -> 1 , py -> 2
depois: px -> 2 , py -> 1

Ponteiro para ponteiro: alocar e "devolver"

Outro uso clássico: a função cria algo (aloca) e precisa entregar o endereço ao chamador. Passamos char **saida e escrevemos o ponteiro *saida = malloc(...) — quem chama recebe a string (e é responsável pelo free).

É o mesmo padrão visto no T2.01 com int ** e claramente com o strtok do T2.04? Não — strtok guarda estado interno; aqui é "devolver alocação por parâmetro", padrão comum em bibliotecas (ex.: ler uma linha).

Código — criar-string.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void criar_saudacao(char **saida)
{
    *saida = malloc(64);
    if (*saida != NULL) {
        strcpy(*saida, "oi, mundo!");
    }
}

int main(void)
{
    char *s = NULL;

    criar_saudacao(&s);
    if (s != NULL) {
        printf("%s\n", s);
        free(s);
    }
    return 0;
}
Saída esperada
oi, mundo!

Exercício 2.11.1

Declare um ponteiro para função int (*op)(int, int), aponte para uma função max que devolve o maior de dois ints, chame através do ponteiro com (7, 2) e (3, 8), e imprima.

Dica: Declaração: int (*op)(int, int) = max; e chamada op(a, b).

Ver solução
#include <stdio.h>

int max(int a, int b)
{
    return a > b ? a : b;
}

int main(void)
{
    int (*op)(int, int) = max;

    printf("max(7, 2) = %d\n", op(7, 2));
    printf("max(3, 8) = %d\n", op(3, 8));
    return 0;
}
Nível 3

Exercício 2.11.2

Ordene um array de Aluno (nome, idade) com qsort pelo campo idade (crescente), usando um comparador que converte void * para const Aluno *. Imprima o resultado.

Dica: qsort(array, n, sizeof(Aluno), cmp_idade); no comparador, const Aluno *a = pa;.

Ver solução
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    char nome[40];
    int idade;
} Aluno;

int cmp_idade(const void *pa, const void *pb)
{
    const Aluno *a = pa;
    const Aluno *b = pb;
    return (a->idade > b->idade) - (a->idade < b->idade);
}

int main(void)
{
    Aluno turma[4] = {
        {"Carla", 33}, {"Ana", 25}, {"Bruno", 30}, {"Diego", 28}
    };
    int n = (int)(sizeof turma / sizeof turma[0]);

    qsort(turma, (size_t)n, sizeof turma[0], cmp_idade);

    for (int i = 0; i < n; i++) {
        printf("%s (%d)\n", turma[i].nome, turma[i].idade);
    }
    return 0;
}

Esperado: Ana(25), Diego(28), Bruno(30), Carla(33).

Nível 3

Exercício 2.11.3

Escreva void aplicar_todos(int (*f)(int), int arr[], int n) que aplica f a cada elemento e imprime o transformado. Teste com uma função dobrar sobre {1,2,3,4}.

Dica: Dentro, for (i...) printf("%d ", f(arr[i])).

Ver solução
#include <stdio.h>

int dobrar(int v) { return v * 2; }

void aplicar_todos(int (*f)(int), int arr[], int n)
{
    for (int i = 0; i < n; i++) {
        printf("%d ", f(arr[i]));
    }
    printf("\n");
}

int main(void)
{
    int v[] = {1, 2, 3, 4};
    aplicar_todos(dobrar, v, 4);
    return 0;
}

Esse é o "map" de linguagens funcionais.

Nível 4

Exercício 2.11.4

Escreva void criar_nome(char **out, const char *nome) que aloca "Olá, NOME!" (use snprintf) e devolve por ponteiro-para-ponteiro. No main, chame, imprima e dê free. Teste com "C" e "C++".

Dica: tamanho = 8 + strlen(nome); alocar; snprintf; *out = s.

Ver solução
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void criar_nome(char **out, const char *nome)
{
    size_t tam = 8 + strlen(nome);
    char *s = malloc(tam);
    if (s == NULL) {
        *out = NULL;
        return;
    }
    snprintf(s, tam, "Ola, %s!", nome);
    *out = s;
}

int main(void)
{
    char *a = NULL, *b = NULL;

    criar_nome(&a, "C");
    criar_nome(&b, "C++");

    if (a != NULL) { printf("%s\n", a); free(a); }
    if (b != NULL) { printf("%s\n", b); free(b); }
    return 0;
}
Nível 4

Exercício 2.11.5

Monte um menu com uma tabela de ponteiros para função: as 4 operações (somar, subtrair, multiplicar, dividir — com divisão checando zero). Leia dois números e a opção do usuário e despache via ops[op](a, b).

Dica: typedef double (*Op)(double, double); e Op ops[4] = {somar, subtrair, multiplicar, dividir};

Ver solução
#include <stdio.h>

typedef double (*Op)(double, double);

double somar(double a, double b)       { return a + b; }
double subtrair(double a, double b)    { return a - b; }
double multiplicar(double a, double b) { return a * b; }
double dividir(double a, double b)     { return a / b; }

int main(void)
{
    Op ops[4] = {somar, subtrair, multiplicar, dividir};
    double a = 20.0, b = 4.0;

    for (int i = 0; i < 4; i++) {
        printf("op[%d]: %.2f\n", i, ops[i](a, b));
    }
    return 0;
}

Versão com scanf: scanf("%d %lf %lf", &op, &a, &b) e valide o intervalo antes de despachar. Divisão por zero seria NaN/inf — trate antes em produção.
