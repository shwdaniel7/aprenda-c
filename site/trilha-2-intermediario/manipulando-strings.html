2.04

Manipulando strings

string.h: strcpy, strcat, strcmp, strchr, strstr, strtok e snprintf

Nível: Intermediário Duração: ~45 min Pré-requisitos: T2.03

🎯 Objetivos

Dominar o kit de funções de para copiar, concatenar, comparar, buscar e separar strings com segurança, evitando as armadilhas clássicas de buffer.

📖 Leitura guiada

Beej's Guide: Capítulo 7 (seção de funções de string; biblioteca string.h)

King (livro em PDF): Capítulos 13.4–13.6 (limites, funções de string)

Foco da leitura: No Beej, a tabela de funções de string com assinaturas e exemplos de uso. No King, as seções 13.4–13.6 explicam por que strncpy/strncat são mais seguras e como funciona a busca.

Copiar: strcpy e strncpy

strcpy(dest, src) copia a origem inteira (parando no \0); é sua responsabilidade garantir que dest tem espaço. strncpy(dest, src, n) copia no máximo n caracteres — e aqui mora a sutileza: não garante \0 automático, então escrevemos dest[n-1] = '\0' à mão:

Código — strcpy.c
#include <stdio.h>
#include <string.h>

int main(void)
{
    char a[32];
    char b[32];

    strcpy(a, "primeira");
    strncpy(b, "segunda", sizeof b - 1);
    b[sizeof b - 1] = '\0';

    printf("a = \"%s\"\n", a);
    printf("b = \"%s\"\n", b);
    return 0;
}
Saída esperada
a = "primeira"
b = "segunda"

Concatenar: strcat e strncat

strcat(dest, src) acha o \0 de dest e copia src a partir dali. Novo estouro clássico: dest deve ter espaço para o resultado todo. strncat limita quantos caracteres vêm de src — a fórmula do tamanho disponível é sizeof(dest) - strlen(dest) - 1:

Código — strcat.c
#include <stdio.h>
#include <string.h>

int main(void)
{
    char frase[64] = "Aprender";
    char resto[32];

    strncat(frase, " C", sizeof frase - strlen(frase) - 1);
    strcpy(resto, " e' divertido");
    strncat(frase, resto, sizeof frase - strlen(frase) - 1);

    printf("%s\n", frase);
    return 0;
}
Saída esperada
Aprender C e' divertido

Comparar: strcmp e strncmp

strcmp(a, b) compara byte a byte e devolve um valor: negativo se a < b, zero se iguais, positivo se a > b (comparação lexicográfica, com os códigos ASCII). strncmp compara só os primeiros n caracteres.

Nunca compare strings com ==: isso compara endereços, não conteúdo.

Código — strcmp.c
#include <stdio.h>
#include <string.h>

int main(void)
{
    printf("strcmp(\"abc\", \"abc\") = %d\n", strcmp("abc", "abc"));
    printf("strcmp(\"abc\", \"abd\") = %d\n", strcmp("abc", "abd"));
    printf("strcmp(\"abd\", \"abc\") = %d\n", strcmp("abd", "abc"));
    printf("strncmp(\"abcde\", \"abcfg\", 3) = %d\n", strncmp("abcde", "abcfg", 3));
    return 0;
}
Saída esperada
strcmp("abc", "abc") = 0
strcmp("abc", "abd") = -1
strcmp("abd", "abc") = 1
strncmp("abcde", "abcfg", 3) = 0

Buscar: strchr e strstr

strchr(s, c) acha o primeiro c na string e devolve um ponteiro para ele (ou NULL). strstr(s, sub) acha a primeira ocorrência de uma substring. Subtrair o ponteiro devolvido do início da string dá o índice:

Código — strchr-strstr.c
#include <stdio.h>
#include <string.h>

int main(void)
{
    const char *texto = "O rato roeu a roupa do rei";
    char *achado;

    achado = strchr(texto, 'r');
    if (achado != NULL) {
        printf("primeiro 'r' no indice %td\n", achado - texto);
    }

    achado = strstr(texto, "roupa");
    if (achado != NULL) {
        printf("\"roupa\" comeca no indice %td\n", achado - texto);
    }
    return 0;
}
Saída esperada
primeiro 'r' no indice 2
"roupa" comeca no indice 14

Separar: strtok (e seus cuidados)

strtok parte a string em "pedaços" usando delimitadores. Primeira chamada recebe a string e o conjunto de delimitadores; as chamadas seguintes recebem NULL e continuam de onde parou.

Cuidados: (1) destrói a string original (insere \0 nos divisores) — passe uma cópia; (2) não é reentrante — não use dentro de dois loops aninhados simultâneos; (3) usa estado global, então não há garantias em múltiplas threads.

Código — strtok.c
#include <stdio.h>
#include <string.h>

int main(void)
{
    char linha[] = "nome:email:idade:cidade";
    char *parte;

    parte = strtok(linha, ":");
    while (parte != NULL) {
        printf("-> %s\n", parte);
        parte = strtok(NULL, ":");
    }
    return 0;
}
Saída esperada
-> nome
-> email
-> idade
-> cidade

Formatar com segurança: snprintf

snprintf(buf, tamanho, formato, ...) é o printf para strings: formata direto no buffer, sempre com \0, e devolve quantos caracteres deveriam ter sido escritos. Se o retorno for maior ou igual ao tamanho do buffer, houve truncamento.

Código — snprintf.c
#include <stdio.h>

int main(void)
{
    char buffer[32];
    int escritos;

    escritos = snprintf(buffer, sizeof buffer, "%s tem %d anos", "Leo", 23);
    if (escritos >= (int)sizeof buffer) {
        printf("String foi truncada.\n");
    }
    printf("buffer = \"%s\" (%d caracteres)\n", buffer, escritos);
    return 0;
}
Saída esperada
buffer = "Leo tem 23 anos" (15 caracteres)

✏️ Exercícios

Nível 1

Exercício 2.04.1

Crie um buffer nome_completo[64] e monte "Fulano da Silva" usando strcpy + strcat (ou strncpy/strncat com limites). Imprima o resultado e o tamanho com strlen.

Dica: Comece com strcpy(nome_completo, "Fulano"), depois acrescente " da Silva".

Ver solução
#include <stdio.h>
#include <string.h>

int main(void)
{
    char nome_completo[64];

    strcpy(nome_completo, "Fulano");
    strncat(nome_completo, " da Silva",
            sizeof nome_completo - strlen(nome_completo) - 1);

    printf("%s (len %zu)\n", nome_completo, strlen(nome_completo));
    return 0;
}
Nível 2

Exercício 2.04.2

Escreva uma função que substitui todas as ocorrências de um caractere por outro em uma string modificável, ex.: trocar todos os "a" por "@" em "banana". Imprima antes e depois.

Dica: Percorra com ponteiro e *p = novo quando *p == velho.

Ver solução
#include <stdio.h>

void substituir(char *s, char velho, char novo)
{
    for (; *s != '\0'; s++) {
        if (*s == velho) {
            *s = novo;
        }
    }
}

int main(void)
{
    char fruta[] = "banana";

    printf("antes : %s\n", fruta);
    substituir(fruta, 'a', '@');
    printf("depois: %s\n", fruta);
    return 0;
}

A string precisa ser um array mutável, jamais um literal (T2.03).

Nível 3

Exercício 2.04.3

Conte as palavras de uma frase usando strtok com delimitadores espaço e tab. Teste com "o rato roeu a roupa".

Dica: Delimitadores: " \t". Cada pedaço devolvido não-NULO é uma palavra.

Ver solução
#include <stdio.h>
#include <string.h>

int main(void)
{
    char frase[] = "o rato roeu a roupa";
    int n = 0;

    char *palavra = strtok(frase, " \t");
    while (palavra != NULL) {
        n++;
        palavra = strtok(NULL, " \t");
    }

    printf("palavras: %d\n", n);
    return 0;
}

Lembre-se: strtok modifica a string original — aqui está tudo bem, frase é uma cópia própria.

Nível 3

Exercício 2.04.4

Escreva um programa que usa strstr para achar a posição da primeira ocorrência de "ca" dentro de "cachorro": e imprime o índice; depois repita buscando algo que não existe ("zz") e trate o NULL.

Dica: Se p = strstr(s, sub), o índice é p - s.

Ver solução
#include <stdio.h>
#include <string.h>

int main(void)
{
    const char *frase = "cachorro";
    char *p;

    p = strstr(frase, "ca");
    if (p != NULL) {
        printf("'ca' em %td\n", p - frase);
    }

    p = strstr(frase, "zz");
    if (p == NULL) {
        printf("'zz' nao encontrado\n");
    }
    return 0;
}

Esperado: ca em 0 e mensagem de não encontrado. Note o uso de %td para a diferença de ponteiros (ptrdiff_t).

Nível 4

Exercício 2.04.5

Leia uma frase com fgets, corte a quebra de linha, use strtok para separar em palavras e imprima as palavras em ordem inversa da digitação. Dica: guarde as palavras num array de ponteiros char *palavras[20].

Dica: Cada token do strtok aponta para dentro do buffer; guarde o ponteiro, não copie.

Ver solução
#include <stdio.h>
#include <string.h>

int main(void)
{
    char frase[256];
    char *palavras[20];
    int n = 0;

    printf("Digite uma frase: ");
    if (fgets(frase, sizeof frase, stdin) == NULL) {
        return 1;
    }
    frase[strcspn(frase, "\n")] = '\0';

    char *token = strtok(frase, " ");
    while (token != NULL && n < 20) {
        palavras[n++] = token;
        token = strtok(NULL, " ");
    }

    printf("inverso: ");
    for (int i = n - 1; i >= 0; i--) {
        printf("%s ", palavras[i]);
    }
    printf("\n");
    return 0;
}

Exemplo: "vou aprender c" sai "c aprender vou". O array palavras guarda ponteiros para dentro de frase (que o strtok fatiou).

🏗️ Miniprojeto: Gravador de perfil de usuário

Crie um programa que pergunta nome, e-mail e cidade (fgets), monta com strncpy/strncat/snprintf um "cartão" formatado, remove o @ do nome para gerar um login, e por fim conta palavras no nome completo. Combina copiar, concatenar, buscar e separar.

Critérios de aceite

  • Todos os buffers com limites (strncpy/strncat/snprintf); nenhum estouro possível.
  • Gera campo "login" removendo espaços e vírgulas do nome.
  • Imprime o cartão formatado e a contagem de palavras.
  • Compila com -Wall -Wextra sem avisos.
