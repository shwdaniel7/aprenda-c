O que é uma struct

Uma struct agrupa várias variáveis relacionadas em um único "objeto", que pode ser copiado, passado a funções e armazenado em arrays. A sintaxe declara o molde (tipo) e depois criamos variáveis do tipo.

Código — pessoa.c
#include <stdio.h>
#include <string.h>

struct Pessoa {
    char nome[50];
    int idade;
    float altura;
};

int main(void)
{
    struct Pessoa p;

    p.idade = 30;
    p.altura = 1.75f;
    strcpy(p.nome, "Marina");

    printf("nome: %s\n", p.nome);
    printf("idade: %d\n", p.idade);
    printf("altura: %.2f\n", p.altura);
    return 0;
}
Saída esperada
nome: Marina
idade: 30
altura: 1.75

Inicializadores e designated

Como arrays, structs podem ser inicializadas por lista {...} na ordem dos campos. C99 adicionou os designated initializers: .campo = valor, que permitem inicializar fora de ordem e deixam o código autoexplicativo.

Código — inicial.c
#include <stdio.h>

struct Pessoa {
    char nome[50];
    int idade;
    float altura;
};

int main(void)
{
    struct Pessoa a = {"Ana", 25, 1.65f};
    struct Pessoa b = {.altura = 1.80f, .idade = 33, .nome = "Bruno"};

    printf("%s, %d anos, %.2f m\n", a.nome, a.idade, a.altura);
    printf("%s, %d anos, %.2f m\n", b.nome, b.idade, b.altura);
    return 0;
}
Saída esperada
Ana, 25 anos, 1.65 m
Bruno, 33 anos, 1.80 m

Passagem por valor vs por ponteiro (->)

Por padrão, uma struct entra na função por cópia: alterar o parâmetro não muda o original. Para modificar o original passamos um ponteiro e usamos o operador -> (que é açúcar para (*p).campo). Structs grandes também são passadas por ponteiro por eficiência.

Código — passagem.c
#include <stdio.h>

struct Pessoa {
    char nome[50];
    int idade;
};

void aniversario_valor(struct Pessoa p)      /* muda a copia */
{
    p.idade++;
}

void aniversario_ponteiro(struct Pessoa *p)  /* muda o original */
{
    p->idade++;
}

int main(void)
{
    struct Pessoa maria = {"Maria", 20};

    aniversario_valor(maria);
    printf("depois de valor:    %d\n", maria.idade);

    aniversario_ponteiro(&maria);
    printf("depois de ponteiro: %d\n", maria.idade);
    return 0;
}
Saída esperada
depois de valor:    20
depois de ponteiro: 21

Cópia com = e arrays de structs

Structs se copiam inteiras com = (campo a campo, incluindo os arrays internos). E assim como int ou char, structs vivem confortavelmente em arrays — o que permite a "tabela" clássica de cadastro:

Código — turma.c
#include <stdio.h>

struct Aluno {
    char nome[40];
    int nota;
};

int main(void)
{
    struct Aluno turma[4] = {
        {"Ana", 8},
        {"Bruno", 6},
        {"Carla", 9},
        {"Diego", 7},
    };
    int melhor = 0;

    for (int i = 1; i < 4; i++) {
        if (turma[i].nota > turma[melhor].nota) {
            melhor = i;
        }
    }

    printf("Melhor nota: %s com %d\n", turma[melhor].nome, turma[melhor].nota);
    return 0;
}
Saída esperada
Melhor nota: Carla com 9

Comparar structs é campo a campo; e o padding

Não existe a == b entre structs (dica: por causa do padding!). A arquitetura alinha campos em fronteiras de 4/8 bytes, deixando "buracos" entre eles — comparar os bytes brutos com memcmp compararia lixo. Por isso comparamos campo a campo.

offsetof (de <stddef.h>) mostra onde cada campo mora, e a ordem dos campos muda o tamanho total:

Código — padding.c
#include <stdio.h>
#include <stddef.h>

struct Compacto {
    char a;
    char b;
    int c;
};

struct Folgado {
    char a;
    int c;
    char b;
};

int main(void)
{
    printf("sizeof(struct Compacto) = %zu\n", sizeof(struct Compacto));
    printf("offsetof a=%zu b=%zu c=%zu\n",
           offsetof(struct Compacto, a),
           offsetof(struct Compacto, b),
           offsetof(struct Compacto, c));

    printf("sizeof(struct Folgado) = %zu\n", sizeof(struct Folgado));
    printf("offsetof a=%zu c=%zu b=%zu\n",
           offsetof(struct Folgado, a),
           offsetof(struct Folgado, c),
           offsetof(struct Folgado, b));
    return 0;
}
Saída esperada
sizeof(struct Compacto) = 8
offsetof a=0 b=1 c=4
sizeof(struct Folgado) = 12
offsetof a=0 c=4 b=8

Nível 1

Exercício 2.06.1

Declare uma struct Produto com nome[30], preco (double) e estoque (int). Crie duas variáveis inicializadas por lista e imprima todas as informações. Depois baixe o preço da segunda em 10%.

Dica: Acesse com p.preco etc. para modificar um campo, basta atribuir.

Ver solução
#include <stdio.h>

struct Produto {
    char nome[30];
    double preco;
    int estoque;
};

int main(void)
{
    struct Produto a = {"Arroz", 22.50, 40};
    struct Produto b = {"Feijao", 9.90, 15};

    printf("%s: R$ %.2f (estoque %d)\n", a.nome, a.preco, a.estoque);
    printf("%s: R$ %.2f (estoque %d)\n", b.nome, b.preco, b.estoque);

    b.preco *= 0.90;   /* desconto de 10%% */
    printf("com desconto: %s: R$ %.2f\n", b.nome, b.preco);
    return 0;
}
Nível 2

Exercício 2.06.2

Escreva a versão por valor e a versão por ponteiro de uma função aplicar_cupom(struct Produto p/struct Produto *p, double desconto) e mostre no main que a versão por valor não altera o original e a por ponteiro altera.

Dica: No ponteiro use p->preco *= (1 - desconto).

Ver solução
#include <stdio.h>

struct Produto {
    char nome[30];
    double preco;
};

void cupom_valor(struct Produto p, double desconto)
{
    p.preco *= 1.0 - desconto;
}

void cupom_ponteiro(struct Produto *p, double desconto)
{
    p->preco *= 1.0 - desconto;
}

int main(void)
{
    struct Produto camiseta = {"Camiseta", 50.0};

    cupom_valor(camiseta, 0.20);
    printf("apos por valor:   %.2f\n", camiseta.preco);

    cupom_ponteiro(&camiseta, 0.20);
    printf("apos por ponteiro: %.2f\n", camiseta.preco);
    return 0;
}

Esperado: 50.00 (o valor não muda na cópia) e depois 40.00.

Nível 3

Exercício 2.06.3

Crie um array de structs Aluno (nome e nota) e escreva uma função buscar(struct Aluno turma[], int n, const char *nome) que devolve o índice do aluno ou -1 se não existir. Teste procurando "Carla" (deve achar) e "Zeca" (não deve).

Dica: Compare com strcmp.

Ver solução
#include <stdio.h>
#include <string.h>

struct Aluno {
    char nome[40];
    int nota;
};

int buscar(struct Aluno turma[], int n, const char *nome)
{
    for (int i = 0; i < n; i++) {
        if (strcmp(turma[i].nome, nome) == 0) {
            return i;
        }
    }
    return -1;
}

int main(void)
{
    struct Aluno turma[3] = {
        {"Ana", 8},
        {"Bruno", 6},
        {"Carla", 9},
    };

    int i = buscar(turma, 3, "Carla");
    printf("Carla no indice %d\n", i);

    i = buscar(turma, 3, "Zeca");
    printf("Zeca no indice %d\n", i);
    return 0;
}

Esperado: Carla no índice 2 e Zeca no índice -1.

Nível 3

Exercício 2.06.4

Crie uma struct Pessoa com um campo nascimento que é outra struct Data (dia, mes, ano). Inicialize com designated (ex.: nascimento .ano = 1999) e imprima no formato dd/mm/aaaa.

Dica: Acesso aninhado: p.nascimento.ano.

Ver solução
#include <stdio.h>

struct Data {
    int dia;
    int mes;
    int ano;
};

struct Pessoa {
    char nome[40];
    struct Data nascimento;
};

int main(void)
{
    struct Pessoa ana = {
        .nome = "Ana",
        .nascimento = {.dia = 10, .mes = 3, .ano = 1999},
    };

    printf("%s nasceu em %02d/%02d/%04d\n",
           ana.nome, ana.nascimento.dia,
           ana.nascimento.mes, ana.nascimento.ano);
    return 0;
}
Nível 4

Exercício 2.06.5

Ordene um array de Aluno por nota (decrescente) com bubble sort, comparando nota. Imprima o ranking antes e depois. A ordenação deve ser estável no nome em caso de empate.

Dica: Compare vizinhos com if (turma[j].nota < turma[j+1].nota) e troque a struct inteira com uma variável temporária do tipo struct.

Ver solução
#include <stdio.h>

struct Aluno {
    char nome[40];
    int nota;
};

int main(void)
{
    struct Aluno turma[5] = {
        {"Ana", 8}, {"Bruno", 6}, {"Carla", 9},
        {"Diego", 7}, {"Elisa", 8},
    };

    for (int i = 0; i < 5; i++) {
        for (int j = 0; j < 4 - i; j++) {
            if (turma[j].nota < turma[j + 1].nota) {
                struct Aluno tmp = turma[j];
                turma[j] = turma[j + 1];
                turma[j + 1] = tmp;
            }
        }
    }

    printf("ranking\n");
    for (int i = 0; i < 5; i++) {
        printf("%d. %s (%d)\n", i + 1, turma[i].nome, turma[i].nota);
    }
    return 0;
}

A struct inteira é copiada na troca (autorizada pelo =). Saída: Carla(9), Ana(8), Elisa(8), Diego(7), Bruno(6).

🏗️ Miniprojeto: Cadastro de alunos

Construa um mini-cadastro com struct Aluno (nome, nota1, nota2, media) e um array fixo de até 30 alunos. O programa apresenta um menu: (1) incluir aluno (lê nome e notas, calcula média), (2) listar, (3) buscar por nome, (4) melhor e pior média, (5) encerrar. Reuse funções com ponteiros e arrays de structs dos módulos anteriores.

Critérios de aceite

  • Menu funcionando com switch e enum (T2.07 já à frente — pode usar int por enquanto).

  • Busca por nome retornando "não encontrado" quando não existir.
  • Nenhum buffer estourado (fgets + limites).
  • Compila com -Wall -Wextra sem avisos.
