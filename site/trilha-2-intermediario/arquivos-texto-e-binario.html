Arquivos: texto e binário

FILE*, fopen, modos r/w/a, fgets/fscanf, fwrite/fread e fseek

Nível: Intermediário Duração: ~45 min Pré-requisitos: T2.04 e T2.06

🎯 Objetivos

Dominar a persistência em C: abrir e fechar arquivos com FILE*, gravar e ler texto com fprintf/fscanf e fgets, gravar e ler binário com fwrite/fread, navegar com fseek/ftell/rewind — e sair de cada programa com arquivos íntegros e bem fechados.

Por que arquivos: persistência

Tudo o que você imprimiu até aqui morreu no terminal. Arquivos são a memória de longo prazo do programa: o dado gravado hoje pode ser lido amanhã — por outro programa seu, pelo editor de texto ou por um banco de dados. Sem arquivos, agendas, logs, configurações e jogos salvos não existiriam.

O meio de acesso é a struct FILE. Quem "abre" um arquivo é fopen(caminho, modo), que devolve um ponteiro FILE * — ou NULL se algo deu errado (permissão, disco cheio, arquivo inexistente...). Regra de ouro: confira sempre o retorno antes de usar.

Código — primeiro-arquivo.c
#include <stdio.h>

int main(void)
{
    FILE *f = fopen("dados.txt", "w");
    if (f == NULL) {
        printf("nao consegui abrir o arquivo para escrita\n");
        return 1;
    }
    fprintf(f, "Ana 8.5\n");
    fprintf(f, "Bruno 7.0\n");
    fclose(f);

    f = fopen("dados.txt", "r");
    if (f == NULL) {
        printf("nao consegui abrir o arquivo para leitura\n");
        return 1;
    }

    char nome[50];
    double nota;
    while (fscanf(f, "%49s %lf", nome, &nota) == 2) {
        printf("li: %s com nota %.1f\n", nome, nota);
    }
    fclose(f);
    return 0;
}
Saída esperada
> gcc -Wall -Wextra -std=c11 primeiro-arquivo.c -o pa
> .\pa.exe

li: Bruno com nota 7.0

Modos de abertura e texto vs binário

fopen recebe o caminho e o modo: "r" (read), "w" (write — apaga o que havia e recomeça do zero), "a" (append — escreve no fim), e os "plus" "r+"/"w+"/"a+", que abrem para leitura E escrita. Neste módulo os programas criam/leem arquivos na pasta atual.

Nos modos de texto (sem "b"), o C traduz a quebra de linha: você escreve \n e o arquivo guarda \r\n no Windows. Os modos binários ("rb", "wb") copiam os bytes exatamente como estão — essencial para arquivos de dados sem tradução.

Código — modos.c
#include <stdio.h>

int main(void)
{
    FILE *f;

    f = fopen("notas.txt", "w");
    if (f == NULL) {
        return 1;
    }
    fprintf(f, "primeira linha\n");
    fclose(f);

    f = fopen("notas.txt", "a");
    if (f == NULL) {
        return 1;
    }
    fprintf(f, "segunda linha (anexada)\n");
    fclose(f);

    f = fopen("notas.txt", "r");
    if (f == NULL) {
        return 1;
    }
    char linha[200];
    int n = 0;
    while (fgets(linha, sizeof linha, f) != NULL) {
        n++;
        printf("%d: %s", n, linha);
    }
    fclose(f);
    return 0;
}
Saída esperada
> gcc -Wall -Wextra -std=c11 modos.c -o modos
> .\modos.exe

1: primeira linha
2: segunda linha (anexada)

Texto: escrever com formato e ler com o "padrão fgets"

Para texto formatado, o par fprintf/fscanf é o printf/scanf com um FILE * como primeiro argumento. Para ler linhas inteiras — incluindo espaços —, o padrão é fgets: ele lê até a quebra de linha e conserva o \n no buffer. E o truque do loop: enquanto fgets devolver NULL, não chegamos ao fim; quando o arquivo acaba, o retorno vira NULL e o loop encerra sozinho.

Depois do loop, feof(f) diz "terminou normalmente"; ferror(f) indica erro real de I/O (disco, permissão). No dia a dia, esperamos o fim natural e conferimos ferror ao sair.

Código — texto.c
#include <stdio.h>
#include <string.h>

int main(void)
{
    FILE *f = fopen("texto.txt", "w");
    if (f == NULL) {
        return 1;
    }
    fprintf(f, "linha um: aprendendo C\n");
    fprintf(f, "linha dois: arquivos\n");
    fprintf(f, "linha tres: misterio termina em EOF\n");
    fclose(f);

    f = fopen("texto.txt", "r");
    if (f == NULL) {
        return 1;
    }

    char buf[200];
    int contador = 0;
    while (fgets(buf, sizeof buf, f) != NULL) {
        contador++;
        printf("linha %d (%zu bytes): %s", contador, strlen(buf), buf);
    }
    if (ferror(f)) {
        printf("erro de leitura!\n");
    } else {
        printf("fim de arquivo (EOF) apos %d linhas\n", contador);
    }
    fclose(f);
    return 0;
}
Saída esperada
> gcc -Wall -Wextra -std=c11 texto.c -o texto
> .\texto.exe

linha 2 (21 bytes): linha dois: arquivos

Binário: fwrite e fread

Arquivo binário guarda a memória "crua": fwrite(endereco, tamanho_do_elemento, quantidade, f) grava quantidade blocos; fread faz o caminho inverso. Os dois devolvem quantos elementos conseguiram processar — confira esse número contra o esperado.

Cuidado com structs: gravar uma struct direto grava também o padding (T2.06), que muda de compilador para compilador. Um arquivo criado numa máquina pode não ser lido de forma igual noutra. Para formatos duradouros, grava-se campo a campo e embute-se um campo versao no próprio arquivo.

Código — binario.c
#include <stdio.h>

typedef struct {
    int id;
    double valor;
    char nome[20];
    int versao;
} Registro;

int main(void)
{
    Registro dados[3] = {
        {1, 10.50, "maca", 1},
        {2, 3.25, "feijao", 1},
        {3, 99.00, "cafe", 1},
    };
    Registro lido[3];
    FILE *f;

    printf("sizeof(Registro) = %zu bytes\n", sizeof(Registro));

    f = fopen("dados.bin", "wb");
    if (f == NULL) {
        return 1;
    }
    size_t escritos = fwrite(dados, sizeof dados[0], 3, f);
    printf("fwrite gravou %zu de 3 registros\n", escritos);
    fclose(f);

    f = fopen("dados.bin", "rb");
    if (f == NULL) {
        return 1;
    }
    size_t lidos = fread(lido, sizeof lido[0], 3, f);
    printf("fread leu %zu de 3 registros\n", lidos);
    fclose(f);

    for (int i = 0; i < 3; i++) {
        printf("id=%d valor=%.2f nome=%s versao=%d\n",
               lido[i].id, lido[i].valor, lido[i].nome, lido[i].versao);
    }
    return 0;
}
Saída esperada
> gcc -Wall -Wextra -std=c11 binario.c -o binario
> .\binario.exe

Navegando: fseek, rewind e ftell

Todo arquivo aberto tem um "ponteiro de posição" interno. fseek(f, offset, SEEK_SET) pula para o byte offset a partir do início (SEEK_CUR a partir do atual, SEEK_END a partir do fim); ftell(f) diz em qual byte você está e rewind(f) volta ao início.

Em um arquivo de registros de tamanho fixo, o registro N mora no offset N × sizeof(Registro). Com um único fseek, lemos só o registro desejado — sem percorrer os anteriores. É o acesso aleatório que bancos de dados usam por trás dos índices.

Código — fseek.c
#include <stdio.h>

typedef struct {
    int id;
    int valor;
} Registro;

int main(void)
{
    Registro regs[6];
    for (int i = 0; i < 6; i++) {
        regs[i].id = 100 + i;
        regs[i].valor = i * 10;
    }

    FILE *f = fopen("registros.bin", "wb");
    if (f == NULL) {
        return 1;
    }
    fwrite(regs, sizeof regs[0], 6, f);
    fclose(f);

    f = fopen("registros.bin", "rb");
    if (f == NULL) {
        return 1;
    }

    int indice = 4;
    long offset = (long)indice * (long)sizeof(Registro);
    if (fseek(f, offset, SEEK_SET) != 0) {
        printf("falha no fseek\n");
        fclose(f);
        return 1;
    }
    printf("ftell depois do fseek: %ld\n", ftell(f));

    Registro r;
    size_t n = fread(&r, sizeof r, 1, f);
    printf("li %zu registro: id=%d valor=%d\n", n, r.id, r.valor);

    rewind(f);
    n = fread(&r, sizeof r, 1, f);
    printf("com rewind li de novo o registro 0: id=%d\n", r.id);

    fclose(f);
    return 0;
}
Saída esperada
> gcc -Wall -Wextra -std=c11 fseek.c -o fseek
> .\fseek.exe

ftell depois do fseek: 32
li 1 registro: id=104 valor=40
com rewind li de novo o registro 0: id=100

(32 = 4 registros × 8 bytes cada; o registro 104 é o 5º, de índice 4)

Boas práticas: o fluxo de um programa de qualidade

Um programa que mexe com arquivo merece os mesmos cuidados de um que mexe com ponteiros. A lista abaixo é o "checklist" que todo código de arquivo sério segue:

  • Fechar sempre: fclose despeja os buffers no disco. Esquecer pode perder o último pedaço de dados e, em volume, esgotar os descritores do processo.
  • Checar o retorno do fclose: boa parte das falhas de escrita (disco cheio, por exemplo) só aparece no fechamento.
  • Conferir NULL logo após o fopen e conferir o retorno dos fwrite/fread/fprintf.
  • Conferir o modo: "w" destrói o arquivo antigo; abrir para escrita algo que devia ser lido é corrupção em câmara lenta.
  • Fechar cada arquivo assim que terminar de usá-lo, mesmo que o programa siga com outros.
Código — fluxo-seguro.c
#include <stdio.h>

int main(void)
{
    FILE *f = fopen("importante.txt", "w");
    if (f == NULL) {
        printf("erro: nao deu para criar o arquivo\n");
        return 1;
    }

    if (fprintf(f, "dados importantes\n") < 0) {
        printf("erro: falha ao escrever\n");
        fclose(f);
        return 1;
    }

    if (fclose(f) != 0) {
        printf("erro: falha ao fechar (dados podem nao ter ido ao disco)\n");
        return 1;
    }
    printf("arquivo salvo e fechado com sucesso\n");
    return 0;
}
Saída esperada
> gcc -Wall -Wextra -std=c11 fluxo-seguro.c -o fluxo
> .\fluxo.exe

arquivo salvo e fechado com sucesso

Nível 1

Exercício 2.13.1

Escreva um programa que grava 5 inteiros num arquivo numeros.txt com fprintf (um por linha), fecha, reabre para leitura e lê tudo com fscanf somando. Imprima cada valor lido e a soma final.

Dica: O loop while (fscanf(f, "%d", &x) == 1) para sozinho ao chegar no fim do arquivo.

Ver solução
#include <stdio.h>

int main(void)
{
    int valores[5] = {3, 8, 15, 22, 47};

    FILE *f = fopen("numeros.txt", "w");
    if (f == NULL) {
        return 1;
    }
    for (int i = 0; i < 5; i++) {
        fprintf(f, "%d\n", valores[i]);
    }
    fclose(f);

    f = fopen("numeros.txt", "r");
    if (f == NULL) {
        return 1;
    }

    int x, soma = 0, lidos = 0;
    while (fscanf(f, "%d", &x) == 1) {
        soma += x;
        lidos++;
        printf("li: %d\n", x);
    }
    fclose(f);

    printf("lidos=%d soma=%d\n", lidos, soma);
    return 0;
}

Para 3, 8, 15, 22 e 47, a soma esperada é 95.

Nível 2

Exercício 2.13.2

Escreva um programa que cria origem.txt com três linhas, copia para destino.txt linha a linha com fgets/fputs e por fim imprime o conteúdo copiado para provar que a cópia funcionou.

Dica: fgets devolve NULL no fim do arquivo; fputs reescreve a linha sem formato (o \n já vem embutido).

Ver solução
#include <stdio.h>

int main(void)
{
    FILE *origem = fopen("origem.txt", "w");
    if (origem == NULL) {
        return 1;
    }
    fprintf(origem, "primeira linha\n");
    fprintf(origem, "segunda linha\n");
    fprintf(origem, "terceira linha\n");
    fclose(origem);

    origem = fopen("origem.txt", "r");
    if (origem == NULL) {
        return 1;
    }
    FILE *dest = fopen("destino.txt", "w");
    if (dest == NULL) {
        fclose(origem);
        return 1;
    }

    char linha[256];
    int copiadas = 0;
    while (fgets(linha, sizeof linha, origem) != NULL) {
        fputs(linha, dest);
        copiadas++;
    }

    fclose(origem);
    if (fclose(dest) != 0) {
        return 1;
    }
    printf("copiei %d linhas\n", copiadas);

    dest = fopen("destino.txt", "r");
    if (dest == NULL) {
        return 1;
    }
    while (fgets(linha, sizeof linha, dest) != NULL) {
        printf("%s", linha);
    }
    fclose(dest);
    return 0;
}

fputs conserva exatamente o que fgets leu, inclusive o \n — nada de formatar de novo.

Exercício 2.13.3

Implemente um "wc" caseiro: crie um arquivo de texto com duas linhas, reabra para leitura e conte linhas, palavras e bytes. Palavra é qualquer sequência de caracteres que não seja espaço, tab ou quebra de linha.

Dica: Conte palavras dentro de cada linha com uma função simples que alterna os estados "dentro de palavra" e "fora de palavra". Bytes = soma de strlen das linhas lidas.

Ver solução
#include <stdio.h>
#include <string.h>

int contar_palavras(const char *linha)
{
    int n = 0;
    int dentro = 0;
    for (const char *p = linha; *p != '\0'; p++) {
        if (*p == ' ' || *p == '\t' || *p == '\n') {
            dentro = 0;
        } else if (!dentro) {
            dentro = 1;
            n++;
        }
    }
    return n;
}

int main(void)
{
    FILE *f = fopen("texto.txt", "w");
    if (f == NULL) {
        return 1;
    }
    fprintf(f, "o rato roeu a roupa do rei de roma\n");
    fprintf(f, "programacao em c\n");
    fclose(f);

    f = fopen("texto.txt", "r");
    if (f == NULL) {
        return 1;
    }

    char linha[512];
    int linhas = 0, palavras = 0;
    long bytes = 0;
    while (fgets(linha, sizeof linha, f) != NULL) {
        linhas++;
        palavras += contar_palavras(linha);
        bytes += (long)strlen(linha);
    }
    fclose(f);

    printf("linhas:  %d\n", linhas);
    printf("palavras: %d\n", palavras);
    printf("bytes:   %ld\n", bytes);
    return 0;
}

Com o conteúdo do exemplo, o resultado real é linhas 2, palavras 12, bytes 52.

Nível 3

Exercício 2.13.4

Crie a struct Contato (nome[40], telefone[20], idade) e grave um array de 3 contatos com fwrite em contatos.bin (modo "wb"). Depois releia com fread e liste tudo.

Dica: fwrite(agenda, sizeof agenda[0], 3, f); confira que o retorno é 3.

Ver solução
#include <stdio.h>

typedef struct {
    char nome[40];
    char telefone[20];
    int idade;
} Contato;

int main(void)
{
    Contato agenda[3] = {
        {"Ana", "119999-0001", 25},
        {"Bruno", "119999-0002", 31},
        {"Carla", "119999-0003", 44},
    };
    Contato lido[3];
    FILE *f;

    f = fopen("contatos.bin", "wb");
    if (f == NULL) {
        return 1;
    }
    size_t n = fwrite(agenda, sizeof agenda[0], 3, f);
    printf("gravados: %zu\n", n);
    fclose(f);

    f = fopen("contatos.bin", "rb");
    if (f == NULL) {
        return 1;
    }
    n = fread(lido, sizeof lido[0], 3, f);
    printf("lidos: %zu\n", n);
    fclose(f);

    for (int i = 0; i < 3; i++) {
        printf("%s | %s | %d anos\n",
               lido[i].nome, lido[i].telefone, lido[i].idade);
    }
    return 0;
}

O arquivo .bin é ilegível num editor de texto: são bytes puros. É o preço — e a vantagem — do binário.

Nível 4

Exercício 2.13.5

Grave 5 registros da struct Record { int id; long pontos; } em records.bin. Depois, sem ler os anteriores, use fseek para ler direto o 3º registro (índice 2).

Dica: O offset do registro N é N * sizeof(Record). Leia um único elemento com fread.

Ver solução
#include <stdio.h>

typedef struct {
    int id;
    long pontos;
} Record;

int main(void)
{
    Record dados[5];
    for (int i = 0; i < 5; i++) {
        dados[i].id = i + 1;
        dados[i].pontos = (long)(i + 1) * 1000;
    }

    FILE *f = fopen("records.bin", "wb");
    if (f == NULL) {
        return 1;
    }
    fwrite(dados, sizeof dados[0], 5, f);
    fclose(f);

    f = fopen("records.bin", "rb");
    if (f == NULL) {
        return 1;
    }

    int indice = 2;
    if (fseek(f, (long)indice * (long)sizeof(Record), SEEK_SET) != 0) {
        fclose(f);
        return 1;
    }

    Record r;
    if (fread(&r, sizeof r, 1, f) != 1) {
        fclose(f);
        return 1;
    }

    printf("registro %d (o 3o da lista) = id %d, pontos %ld\n",
           indice + 1, r.id, r.pontos);
    fclose(f);
    return 0;
}

Chave do acesso aleatório: registros de tamanho fixo e a conta N × sizeof. Valide o índice antes de pagar o fseek em produção.

🏗️ Miniprojeto: Agenda de contatos binária

Construa uma agenda persistente em agenda.bin com a struct Contato (nome, telefone, idade) gravada via fwrite. O programa abre o arquivo (cria se não existir), lê os registros existentes com fread para um vetor em memória e apresenta um menu: (1) adicionar contato, (2) listar todos, (3) buscar por posição (índice) usando fseek sem ler os anteriores, (4) sair — reescrevendo o vetor inteiro com fwrite antes de fechar.

Critérios de aceite

  • Leitura inicial e gravação final sempre via fwrite/fread sobre a mesma struct; nada de texto.
  • Adicionar respeita um teto definido por você (ex.: 100 contatos) com mensagem clara.
  • Buscar por posição usa fseek com N * sizeof(Contato) e confirma o resultado; índice fora do intervalo é tratado.
  • Todos os fopen conferem NULL e todos os fclose conferem o retorno.
  • Compila com gcc -Wall -Wextra -std=c11 sem avisos e trabalha com o arquivo na pasta atual.
